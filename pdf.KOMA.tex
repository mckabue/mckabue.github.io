% Load necessary packages
\usepackage{scrlayer-scrpage}
\usepackage[dvipsnames]{xcolor}
\usepackage{enumitem}
\usepackage{graphicx}
\usepackage[colorlinks=true, allcolors=blue]{hyperref}

% Set page geometry
\usepackage[
  top=0.65in,
  bottom=0.65in,
  left=0.6in,
  right=0.6in,
  footskip=0pt, % Eliminates the space reserved for the footer
  nofoot % Ensures that no footer is placed at the bottom of the page.
]{geometry}
\pagenumbering{gobble}  % This disables page numbering

% Line spacing — slightly open for readability
\usepackage{setspace}
\setstretch{1.12}

% Configure headers and footers
\clearpairofpagestyles
\cfoot[\pagemark]{\pagemark}

% List breathing room
\setlist{itemsep=3pt, topsep=4pt, parsep=0pt, partopsep=0pt}

% Section and subsection spacing + thin rule under section headings
\usepackage{titlesec}
\titleformat{\section}{\large\bfseries}{}{0em}{}
  [\vspace{3pt}{\color{gray!55}\hrule height 0.4pt}\vspace{1pt}]
\titlespacing*{\section}{0pt}{0.9\baselineskip}{0.4\baselineskip}
\titlespacing*{\subsection}{0pt}{0.65\baselineskip}{0.2\baselineskip}

% Paragraph gap between job blocks
\setlength{\parskip}{3pt}

\definecolor{backgroundColor}{HTML}{edeceb}

\usepackage{tcolorbox}
\newcommand{\borderBox}[1]{%
  \begin{tcolorbox}[
    colback=backgroundColor,
    boxrule=0pt,
    arc=0pt,
    left=0pt,
    right=0pt,
    top=2pt,
    bottom=2pt,
    boxsep=0pt,
    width=\textwidth
  ]
    #1
  \end{tcolorbox}%
}
